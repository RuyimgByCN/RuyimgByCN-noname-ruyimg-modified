447: pub fn run(
448:     core: &mut GameStateCore,
449:     res: &mut Resolution,
450:     catalog: &crate::catalog::Catalog,
451:     out: &mut Vec<LogEvent>,
452:     slice: Option<u32>,
453: ) -> StepOutcome {
454:     let mut steps: u32 = 0;
455:     loop {
456:         if let Some(fault) = res.fault.take() {
457:             res.fault = Some(fault);
458:             return StepOutcome::Fault;
459:         }
460:         if res.budget == 0 {

462:         }
463:         if let Some(limit) = slice {
464:             if steps >= limit {
465:                 return StepOutcome::Yielded;

467:         }
468:         steps += 1;

472:         for frame in &mut res.stack {

474:                 if d.source.is_some_and(|source| !core.player(source).alive) {
475:                     d.source = None;
476:                 }
477:             }
478:         }
479: 
480:         // A) 恢复深度匹配的挂起机器
481:         if res.open_choice.is_none() {

483:                 let mut park = res.parks.remove(idx);
484:                 let prior_skills: std::collections::BTreeSet<_> =
485:                     core.skills.keys().copied().collect();
486:                 let flow = {
487:                     let mut ctx = Ctx {
488:                         core,
489:                         res,
490:                         catalog,
491:                         out,
492:                     };
493:                     park.machine.step(&mut ctx)
494:                 };
495:                 notify_new_skills(core, res, catalog, &prior_skills);
496:                 res.answer = None; // 应答只消费一次
497:                 match flow {
498:                     Flow::Done => finish_slot(res, park.slot),
499:                     Flow::Yield => {
500:                         if park.machine.trigger_order().is_some() {
501:                             res.parks.insert(0, park);
502:                         } else {
503:                             res.parks.push(park);
504:                         }
505:                     }
506:                     Flow::WaitChoice { observer, spec } => {
507:                         let id = res.next_choice_id;
508:                         res.next_choice_id += 1;
509:                         out.push(LogEvent::ChoiceOpened {
510:                             id,
511:                             observer,
512:                             prompt: spec.prompt,
513:                         });
514:                         res.open_choice = Some(OpenChoice { id, observer, spec });
515:                         // 应答恢复点＝**开窗时刻**的栈深。机器可能在挂起期间
516:                         // 发射过子帧（如 AOE 逐目标派发：前一目标的伤害帧
517:                         // 尚未结算就为下一目标开响应窗），此时沿用创建时的
518:                         // 旧深度会与其它挂起机器错配——应答被深度恰好相等
519:                         // 的**别的**机器消费（AOE→昏迷应答错配：目标机器
520:                         // 恢复时 answer 已空，panic 实证）。开窗后对局暂停
521:                         // （open_choice 存在时不推进栈），故开窗时刻的栈深
522:                         // 即恢复点。
523:                         park.resume_depth = res.stack.len();
524:                         res.parks.push(park);
525:                         return StepOutcome::Pending;
526:                     }
527:                 }
528:                 continue;
529:             }
530:         }
